# Navigation

cd /var/log - Navigate to logs in linux
ls -lah - List all files and permissions
pwd - Print working directory

# File operations
touch test.txt - Create text file
mkdir -p projects/demo - Make directory
cp test.txt projects/demo/ - Copy file to directory
mv projects/demo/test.txt projects/demo/backup.txt - Rename directory
rm projects/demo/backup.txt - Delete directory

# Viewing files
cat /etc/passwd - View file
less /var/log/syslog - 
head -n 20 /etc/service - View first 20 lines of a file
tail -f /var/log/auth.log - View last 20 lines of a file
